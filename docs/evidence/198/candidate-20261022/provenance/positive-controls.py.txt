import subprocess,json,os,time
from pathlib import Path
p=Path('.tmp/e15-readiness-20261022').resolve();out=p/'positive-controls';out.mkdir(exist_ok=False)
k=['kubectl','--kubeconfig',str(p/'kubeconfig'),'--context','kind-agenova-e15-strict-20261022'];env=dict(os.environ,KUBECONFIG=str(p/'kubeconfig'));cli=[str(p/'agenova'),'--state-dir',str(p/'state')]
def modelcount(): return sum('[GIN]' in l and '/v1/chat/completions' in l for l in (Path.home()/'.ollama/logs/server.log').read_text().splitlines())
def mcplog(): return subprocess.check_output(k+['-n','agenova-e16','logs','deploy/e16-mcp'],text=True)
def request(ref,obj,tools,scopes):return {'apiVersion':'agenova.io/v1alpha1','kind':'ClaimRequest','metadata':{'name':ref},'spec':{'templateRef':'e15-engineer','projectRef':'payments','task':{'type':'investigation','input':{'objective':obj}},'requestedAccess':{'tools':tools,'resourceScopes':scopes,'modelProfile':'coding-standard'},'runtime':{'profileRef':'standard-isolated','timeout':'3m'}}}
rows=[]
for scene,body in [('model',request('e15-ready-model-20261022','e15-scenario=control',[],[])),('tool',request('e15-ready-tool-20261022','e15-scenario=reference Read README.md in repo:e15/work-a and summarize it in one sentence.',['repo.read'],['repo:e15/work-a']))]:
 watches=[]
 for resource,base in [('pods','/api/v1'),('sandboxclaims','/apis/extensions.agents.x-k8s.io/v1alpha1')]:
  url=base+'/namespaces/agenova-e15-ready-20261022/'+resource
  snap=json.loads(subprocess.check_output(k+['get','--raw',url]));rv=snap['metadata']['resourceVersion']
  f=(out/(scene+'.'+resource+'.watch.jsonstream')).open('w');err=(out/(scene+'.'+resource+'.watch.stderr')).open('w');proc=subprocess.Popen(k+['get','--raw',url+'?watch=true&resourceVersion='+rv+'&timeoutSeconds=300'],stdout=f,stderr=err);watches.append((resource,proc,f,err))
 time.sleep(1);before={'model':modelcount(),'mcp':mcplog().count('"event":"tool"')};req=out/(scene+'.request.json');req.write_text(json.dumps(body,indent=2)+'\n')
 try:
  r=subprocess.run(cli+['run','-f',str(req),'--json'],env=env,text=True,capture_output=True,timeout=230)
  (out/(scene+'.run.json')).write_text(r.stdout);(out/(scene+'.run.stderr')).write_text(r.stderr)
  time.sleep(2);after={'model':modelcount(),'mcp':mcplog().count('"event":"tool"')}
 finally:
  for _,proc,f,err in watches:proc.terminate();proc.wait();f.close();err.close()
 view=json.loads(r.stdout) if r.stdout.strip() else {};facts=view.get('facts',[]);state=view.get('state',{});events={}
 for resource,_,_,_ in watches:
  es=[json.loads(l) for l in (out/(scene+'.'+resource+'.watch.jsonstream')).read_text().splitlines() if l.strip()];events[resource]={t:sum(e.get('type')==t for e in es) for t in ['ADDED','MODIFIED','DELETED','ERROR']}
 row={'scenario':scene,'exit':r.returncode,'before':before,'after':after,'delta':{s:after[s]-before[s] for s in before},'phase':state.get('claim',{}).get('phase'),'outcome':view.get('outcome'),'providerAttempts':{op:sum(f.get('kind')=='ProviderAttempt' and f.get('operation')==op for f in facts) for op in ['model.invoke','tool.invoke']},'watchEvents':events,'toolInvocationIds':[f.get('invocationId') for f in facts if f.get('kind')=='ToolDecision' and f.get('result')=='Allow']}
 rows.append(row);(out/'summary.json').write_text(json.dumps(rows,indent=2)+'\n');(out/'mcp-receipts.log').write_text(mcplog());print(json.dumps(row),flush=True)
 if row['phase']!='Succeeded':raise SystemExit('Positive control failed; stop without further submission')
 if scene=='model' and (row['delta']['model']<=0 or row['delta']['model']!=row['providerAttempts']['model.invoke']):raise SystemExit('Model oracle mismatch; stop')
 if scene=='tool' and (row['delta']['mcp']<=0 or row['delta']['mcp']!=row['providerAttempts']['tool.invoke']):raise SystemExit('MCP oracle mismatch; stop')
